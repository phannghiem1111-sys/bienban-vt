function api(fn) {
  var args = Array.prototype.slice.call(arguments, 1);
  return new Promise(function (resolve, reject) {
    if (!(window.google && google.script && google.script.run)) {
      if (window.MOCK) return MOCK(fn, args).then(resolve, reject);
      if (!API_URL) return reject(new Error('Không kết nối được máy chủ.'));
      return fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ fn: fn, args: args }) })
        .then(function (r) { return r.json(); })
        .then(function (j) {
          if (j && j.ok) return resolve(j.data);
          var msg = String((j && j.error) || 'Lỗi máy chủ.');
          if (msg.indexOf('PHIEN_HET_HAN') >= 0) { logoutLocal(); return reject(new Error('Phiên đăng nhập hết hạn, đăng nhập lại.')); }
          reject(new Error(msg));
        }, function () { reject(new Error('Không kết nối được máy chủ. Kiểm tra mạng rồi thử lại.')); });
    }
    var r = google.script.run.withSuccessHandler(resolve).withFailureHandler(function (e) {
      if (String(e && e.message).indexOf('PHIEN_HET_HAN') >= 0) { logoutLocal(); reject(new Error('Phiên đăng nhập hết hạn, đăng nhập lại.')); }
      else reject(e);
    });
    r[fn].apply(r, args);
  });
}

function autocomplete(input, finder, render, onPick) {
  var wrap = input.parentNode, dd = document.createElement('div'); dd.className = 'dd hidden'; wrap.appendChild(dd);
  var items = [], idx = -1;
  function show() {
    items = finder(input.value); idx = -1;

    dd.innerHTML = items.map(function (it, i) { return '<div class="opt" data-i="' + i + '">' + render(it) + '</div>'; }).join('');
    dd.classList.remove('hidden');
  }
  input.addEventListener('input', show);
  input.addEventListener('focus', function () { if (input.value) show(); });
  input.addEventListener('keydown', function (e) {
    var opts = $$('.opt[data-i]', dd);
    if (e.key === 'ArrowDown') { idx = Math.min(idx + 1, opts.length - 1); }
    else if (e.key === 'ArrowUp') { idx = Math.max(idx - 1, 0); }
    else if (e.key === 'Enter' && idx >= 0) { e.preventDefault(); pick(idx); return; }
    else return;
    e.preventDefault(); opts.forEach(function (o, i) { o.classList.toggle('on', i === idx); });
  });
  dd.addEventListener('mousedown', function (e) { var o = e.target.closest('.opt[data-i]'); if (o) { e.preventDefault(); pick(+o.dataset.i); } });
  input.addEventListener('blur', function () { setTimeout(function () { dd.classList.add('hidden'); }, 150); });
  function pick(i) { dd.classList.add('hidden'); onPick(items[i]); }
}

function toCanvas(img, max, crop) {
  var sx = 0, sy = 0, sw = img.naturalWidth, sh = img.naturalHeight;
  if (crop) { sx = sw * crop[0]; sy = sh * crop[1]; sw = sw * crop[2]; sh = sh * crop[3]; }
  var k = Math.min(1, max / Math.max(sw, sh)), c = document.createElement('canvas');
  c.width = Math.round(sw * k); c.height = Math.round(sh * k);
  c.getContext('2d').drawImage(img, sx, sy, sw, sh, 0, 0, c.width, c.height);
  return c;
}

/** đọc QR trong ảnh: thử cả ảnh, rồi các góc (QR in ở góc trên phải) */
function readQr(img) {
  if (!window.jsQR) return '';
  var tries = [[1400, null], [1800, [0.5, 0, 0.5, 0.45]], [1800, [0.4, 0, 0.6, 0.6]], [1000, null], [1800, [0, 0, 1, 0.5]]];
  for (var i = 0; i < tries.length; i++) {
    var c = toCanvas(img, tries[i][0], tries[i][1]), d = c.getContext('2d').getImageData(0, 0, c.width, c.height);
    var r = jsQR(d.data, c.width, c.height, { inversionAttempts: 'attemptBoth' });
    if (r && r.data) return r.data.trim();
  }
  return '';
}
/* QR dạng ma trận '0'/'1' (có viền trắng 2 ô) để server vẽ bằng bảng HTML, vì bộ tạo PDF của Google không hiện ảnh data: */
function qrMatrix(text) {
  var q = qrcode(0, 'M'); q.addData(text); q.make();
  var n = q.getModuleCount(), pad = 2, rows = [], blank = new Array(n + 2 * pad + 1).join('0');
  for (var k = 0; k < pad; k++) rows.push(blank);
  for (var r = 0; r < n; r++) { var s = ''; for (var c = 0; c < n; c++) s += q.isDark(r, c) ? '1' : '0'; rows.push('00' + s + '00'); }
  for (k = 0; k < pad; k++) rows.push(blank);
  return rows;
}

/* Ngữ cảnh hộp xuất Excel cho màn Tra tồn kho. Dữ liệu do máy chủ (xuatTon) giới hạn sẵn theo quyền: Quản trị / Trưởng phòng / Trưởng BP toàn bộ, Đội trưởng cả đội, nhân viên chỉ của mình. */
function tonCtx(d) {
  var self = d.loai === 'self', nguoi = d.nguoi.filter(function (p) { return p.items.length; }), byMa = {}, order = [];
  nguoi.forEach(function (p) {
    p.gt = 0; p.thieu = 0;
    p.items.forEach(function (it) {
      var x = byMa[it[0]];
      if (!x) { x = byMa[it[0]] = { ma: it[0], ten: it[1], dvt: it[2], sl: 0, gia: it[4] || 0, ai: [] }; order.push(x); }
      x.sl += it[3]; x.ai.push((p.hoTen || p.ten) + ' ' + num(it[3]));
      if (it[4]) p.gt += it[3] * it[4]; else p.thieu++;
    });
  });
  nguoi.sort(function (a, b) { return b.gt - a.gt || b.items.length - a.items.length; });
  order.sort(function (a, b) { return b.sl * b.gia - a.sl * a.gia || b.sl - a.sl; });
  return {
    loai: d.loai, coGia: !!d.coGia,
    note: 'Phạm vi của bạn: ' + d.nhan,
    scope: function () { return d.loai === 'team' ? d.nhan : ''; },
    tens: function () { return []; }, // Tra VT: máy chủ tự giới hạn theo quyền
    nv: function () {
      return { n: nguoi.length,
        rows: nguoi.map(function (p) { return [p.hoTen || p.ten, p.ten, p.mien || '', p.doiTruong || '', p.items.length, Math.round(p.gt), p.thieu || '']; }),
        detail: [].concat.apply([], nguoi.map(function (p) { return p.items.map(function (it) { return [p.hoTen || p.ten, it[0], it[1], it[2], it[3], it[4] || '', it[4] ? Math.round(it[3] * it[4]) : '']; }); })) };
    },
    vt: function () {
      return { n: order.length, holders: !self,
        rows: order.map(function (x) { return [x.ma, x.ten, x.dvt, Math.round(x.sl * 100) / 100, x.gia || '', x.gia ? Math.round(x.sl * x.gia) : '', x.ai.join('; ')]; }) };
    }
  };
}

      });
    };

    draw();
  }).catch(function (e) { m.innerHTML = '<div class="msg err">' + esc(e.message) + '</div>'; });
}

        busy(true, 'Đang điều chuyển…');
        api('dieuChuyen', S.token, A, B, dong).then(function (r) { busy(false); toast('Đã điều chuyển ' + r.soLoai + ' loại (' + r.id + ')'); viewDC(); })
          .catch(function (e) { busy(false); var t = (e && e.message) || String(e); toast(/apply|not a function|undefined/i.test(t) ? 'Máy chủ chưa có chức năng này. Cần dán Code.gs bản mới rồi Deploy phiên bản mới.' : t, 'err'); });
      });
    };
    drawB(); drawBody(); drawHist();
  }).catch(function (e) { m.innerHTML = '<div class="msg err">' + esc(e.message) + '</div>'; });

    busy(true, 'Đang tạo file PDF ' + p.soBB + '…');

/* Upload bản ký: chụp → đọc QR trong ảnh → so với số BB + phiên bản */
function uploadSigned(b) {
  pickImage().then(loadImg).then(function (img) {
    busy(true, 'Đang đọc mã QR trên biên bản…');
    setTimeout(function () {
      var qr = readQr(img), expect = b.soBB + '|v' + b.phienBan;
      if (qr && qr.split('|')[0] !== b.soBB) { busy(false); return toast('Ảnh là biên bản khác (' + qr.split('|')[0] + '). Chụp đúng tờ ' + b.soBB + '.', 'err'); }
      if (qr && qr !== expect) { busy(false); return toast('Ảnh là bản in cũ (' + qr.split('|')[1] + '). Bản hợp lệ là v' + b.phienBan + '.', 'err'); }
      var send = function () {
        busy(true, 'Đang tải bản ký lên…');
        api('uploadSigned', S.token, b.id, compress(img, 2000), qr).then(function (nb) { busy(false); toast('Đã upload bản ký. Chờ đội trưởng duyệt.'); viewDetail(nb.id, nb); }).catch(fail);
      };
      if (!qr) { busy(false); confirmBox('Không đọc được mã QR', 'Ảnh có thể bị mờ hoặc thiếu góc có mã QR. Nên chụp lại rõ cả tờ. Vẫn gửi ảnh này?', 'Vẫn gửi', send); }
      else send();
    }, 50);
  }).catch(fail);
}

function approveBox(b) {
  var o = $('#overlay'), pick = b.deXuat || 'Xuất kho';
  o.innerHTML = '<div class="box"><h3 style="margin-top:0">Duyệt ' + esc(b.soBB) + '</h3>' +
    '<div class="muted">NV đề xuất: <b>' + esc(b.deXuat) + '</b>. ' + b.vatTu.length + ' khoản vật tư.</div>' +
    '<label>Kết quả</label><div class="chips" id="apK">' + ['Xuất kho', 'Mua mới', 'Bảo hành', 'Trả lại'].map(function (k) { return '<button class="chip' + (k === pick ? ' on' : '') + '" data-v="' + k + '">' + k + '</button>'; }).join('') + '</div>' +
    '<div class="muted" id="apHint" style="margin-top:8px"></div>' +
    '<label>Ghi chú <span id="apReq" class="req hidden">*</span></label><textarea id="apNote" placeholder="Lý do trả lại / ghi chú"></textarea>' +
    '<div class="row" style="margin-top:12px"><button class="btn" id="apNo">Hủy</button><button class="btn ok" id="apOk">Xác nhận</button></div></div>';
  o.classList.remove('hidden'); o.onclick = null;
  function hint() {
    $('#apHint').textContent = pick === 'Xuất kho' ? 'Ghi ' + b.vatTu.length + ' dòng vào sheet Tra VT.' : pick === 'Trả lại' ? 'Biên bản về Nháp, NV sửa và in lại phiên bản mới.' : 'Không ghi vào Tra VT.';
    $('#apReq').classList.toggle('hidden', pick !== 'Trả lại');
  }
  hint();
  $$('#apK .chip').forEach(function (c) { c.onclick = function () { pick = c.dataset.v; $$('#apK .chip').forEach(function (x) { x.classList.toggle('on', x === c); }); hint(); }; });
  $('#apNo').onclick = function () { o.classList.add('hidden'); };
  $('#apOk').onclick = function () {
    var note = $('#apNote').value.trim();
    if (pick === 'Trả lại' && !note) return toast('Nhập lý do trả lại.', 'err');
    busy(true, 'Đang duyệt…');
    api('approve', S.token, b.id, pick, note).then(function (nb) { busy(false); toast(pick === 'Trả lại' ? 'Đã trả lại cho NV.' : 'Đã duyệt: ' + pick); viewDetail(nb.id, nb); }).catch(fail);
  };
}

/* ======================= QUÉT QR ======================= */
function viewScan() {
  setTitle('Quét QR biên bản', false);
  $('#main').innerHTML = '<div class="card" style="text-align:center"><div style="font-size:54px;line-height:1.2">⌗</div>' +
    '<p>Chụp ảnh tờ biên bản <b>đã ký, đóng dấu</b>. App đọc mã QR, mở đúng biên bản và tải bản ký lên.</p>' +
    '<button class="btn pri block" id="scBtn">Chụp biên bản</button>' +
    '<p class="muted">Mẹo: chụp thẳng, đủ 4 góc, không bị lóa ở góc có mã QR.</p></div>';
  $('#scBtn').onclick = function () {
    var image;
    pickImage().then(loadImg).then(function (img) {
      image = img; busy(true, 'Đang đọc mã QR…');
      return new Promise(function (r) { setTimeout(function () { r(readQr(img)); }, 50); });
    }).then(function (qr) {
      if (!qr) { busy(false); return toast('Không đọc được mã QR. Chụp lại rõ hơn, hoặc mở biên bản trong danh sách rồi bấm Upload bản ký.', 'err'); }
      return api('findByQr', S.token, qr).then(function (r) {
        if (!r.khopPhienBan) { busy(false); return toast('Đây là bản in cũ v' + r.phienBanQr + '. Bản hợp lệ là v' + r.phienBanHienTai + '.', 'err'); }
        return api('getBB', S.token, r.id).then(function (b) {
          if (b.maNVLap !== S.user.maNV) { busy(false); push(viewScan); return viewDetail(b.id, b); }
          busy(true, 'Đang tải bản ký lên…');
          return api('uploadSigned', S.token, b.id, compress(image, 2000), qr).then(function (nb) { busy(false); toast('Đã upload bản ký ' + nb.soBB + '.'); push(viewScan); viewDetail(nb.id, nb); });
        });
      });
    }).catch(fail);
  };
}

/* nhận biết khung quá thấp so với màn hình điện thoại */
function fitFrame() {
  var ae = document.activeElement;
  if (ae && /^(INPUT|TEXTAREA|SELECT)$/.test(ae.tagName)) return; // bàn phím đang mở làm khung thấp đi, không đổi bố cục
  var short = window.innerWidth < 900 && screen.height > 0 && window.innerHeight < screen.height * 0.72;
  document.body.classList.toggle('shortframe', short);
  var tb = document.querySelector('.topbar');
  if (tb) document.documentElement.style.setProperty('--tbh', tb.offsetHeight + 'px');
}
window.addEventListener('resize', fitFrame);
setInterval(fitFrame, 1500);
